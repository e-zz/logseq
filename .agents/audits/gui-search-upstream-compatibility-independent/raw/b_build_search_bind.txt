(defn- build-search-bind
  [q input page block limit use-namespace-last-part?]
  (let [namespace? (and use-namespace-last-part?
                        (ns-util/namespace-page? q))
        last-part (when namespace?
                    (some-> (text/get-namespace-last-part q)
                            get-match-input))]
    (cond-> (cond-> [] page (conj page) block (conj block))
      true (conj input)
      namespace? (conj last-part)
      true (conj limit))))
